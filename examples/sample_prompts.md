## Usage

```python
from engine import festive, showroom, delivery, generate, generate_json

# Quick one-liner
print(festive(car_model="Nexon", brand="Tata", festival="Diwali"))

# Full custom
from engine import VeoParams, Category, Tone, CameraMove, ShotType, Character

params = VeoParams(
    category=Category.CAR_DELIVERY,
    brand="Hyundai",
    car_model="Creta",
    car_color="abyss black",
    tone=Tone.WARM,
    shot_type=ShotType.MEDIUM,
    camera_move=CameraMove.TRACKING,
    dialogue_char=Character(
        description="the proud owner",
        voice_style="emotional, proud",
        dialogue="This is our dream car",
    ),
)
print(generate(params))        # Full prompt
print(generate_json(params))   # Dict for API
```
